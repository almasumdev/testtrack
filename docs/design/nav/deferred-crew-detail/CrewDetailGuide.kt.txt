        onDismiss = onDismiss,

        fitMax = 0.9f,

        Column(
            Modifier.weight(1f, fill = false).verticalScroll(rememberScrollState()).padding(bottom = 20.dp),
            verticalArrangement = Arrangement.spacedBy(24.dp),
        ) {
            CrewGuideGroup(Lucide.Users, "Before the first night") {
                CrewGuideBullet(
                    if (crew.firstNight > 1) {
                        "This crew already started. Your first night begins at the first local midnight after the team puts your app in."
                    } else {
                        "All ${CrewRules.SEATS} apps must be confirmed. Your run starts at your first local midnight after the crew fills."
                    },
                )
                CrewGuideBullet(
                    "Install the other apps from Google Play using the account in the testers group. Tap an app tile for its test sign-in.",
                )
                if (crew.firstNight > 1) {
                    CrewGuideBullet(
                        "You join on crew day ${crew.firstNight}: ${crew.ownNights} nights for you. Earlier nights are not owed.",
                    )
                } else {
                    CrewGuideBullet("This crew runs for ${crew.runDays} nights. Each day ends at your local midnight.")
                }
                CrewGuideBullet(
                    "If Play cannot check an app, members try it: ${CrewRules.VOTES_TO_DECIDE} Yes answers confirm it; ${CrewRules.VOTES_TO_DECIDE} No answers send it back.",

            }

                    "Use Open all, or open apps one at a time. Each stays open for ${CrewRules.OPEN_SECONDS} seconds; at least ${CrewRules.VISIT_SECONDS} seconds in front and a picture are needed to count.",

                CrewGuideBullet(
                    "The floating timer moves to the next app after ${CrewRules.OPEN_SECONDS} seconds. Pause it when you need longer.",

                    "Only apps listed in Tonight are owed. Optional tonight is not required; apps under Left the run are not owed.",
                )
            }
            CrewGuideGroup(Lucide.Camera, "Pictures and sending") {
                CrewGuideBullet(
                    "TestTrack takes one picture while the app is in front. It counts once time and the picture reach the server.",
                )
                CrewGuideBullet(
                    "Pictures go to your own Google Drive, shared by link. A pending upload retries by itself and keeps its original night, even after midnight.",
                )
                CrewGuideBullet(
                    "Picture missing? Open the app again, or pick a picture from your phone. A picked picture is marked as added by hand.",
                )
                CrewGuideBullet("An upload waiting more than 3 hours is highlighted. Check your connection and use Try again.")
            }
            CrewGuideGroup(Lucide.MessageSquare, "Private feedback on Play") {
                CrewGuideBullet(
                    "Send useful private feedback once per app: what worked, what was confusing, and what could improve. Crew feedback earns no tix.",
                )
                CrewGuideBullet(
                    "Send it by your night ${CrewRules.FEEDBACK_BY_NIGHT} with that app, with screenshot proof. After that, its opens stop counting until feedback is sent.",
                )
                CrewGuideBullet(
                    "Use Feedback on the app's row, or open the row for its actions. The developer reads it in Play Console; the team can see the proof.",
                )
            }
            CrewGuideGroup(Lucide.RefreshCw, "Updates and app problems") {
                CrewGuideBullet(
                    "An old copy counts for ${CrewRules.UPDATE_GRACE_HOURS} hours after the owner announces an update. After that, update from Play before opening it again.",
                )
                CrewGuideBullet(
                    "Your own Update control announces the newer Play version installed on this phone. Down tonight excuses your app for that night; you have ${CrewRules.DOWN_NIGHTS} per run.",
                )
                CrewGuideBullet(
                    "An app does not work? Open its tile and report what happened. During a run, a report excuses that app for that night only.",
                )

            CrewGuideGroup(Lucide.TriangleAlert, "Missed nights and getting back in") {

                    "A night is missed when ${CrewRules.MISS_UNOPENED} or more owed apps stay unopened, or you open none. One or two left still lose a clean-run refund.",

                CrewGuideBullet(

                CrewGuideBullet(

        }
    }

@Composable
private fun CrewGuideGroup(
    icon: ImageVector,
    title: String,
    content: @Composable () -> Unit,
) {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        ReviewSectionTitle(icon, title)
        content()
    }
}

/** Same dot and inset as the feedback guide; rule figures and key actions catch the eye. */
@Composable
internal fun CrewGuideBullet(
    body: String,
    tone: Tone? = null,
) {
    val ink = tone?.ink()
    val strong = SpanStyle(fontWeight = FontWeight.SemiBold, color = ink ?: Tt.colors.text)
    val words =
        Regex(
            "\\b\\d+(?:\\s+(?:seconds|hours|nights|tix))?\\b|Google Play|Google Drive|Open all|Tonight|Optional tonight|Left the run|Try again|Ask to get it back",
        )
    Box(Modifier.padding(start = 12.dp)) {
        RuleLine(
            buildAnnotatedString {
                append(body)
                if (ink != null) addStyle(SpanStyle(color = ink), 0, length)
                words.findAll(body).forEach { addStyle(strong, it.range.first, it.range.last + 1) }
            },
        )
    }
}
