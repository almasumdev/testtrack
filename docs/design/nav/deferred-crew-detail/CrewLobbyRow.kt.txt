/** A crew's fifteen places first, the reader's app last. Opening the card still opens its crew. */
@Composable
internal fun CrewLobbyRow(
    crew: Crew,
    onOpen: () -> Unit,
) {
    val mine = crew.mine
    CrewCard(padding = PaddingValues(16.dp), onClick = onOpen) {
        Row(verticalAlignment = Alignment.CenterVertically) {
            Text(crew.name, Modifier.weight(1f), style = tileStyle(18f, 23f, FontWeight.SemiBold), color = Tt.colors.text)
            Spacer(Modifier.width(8.dp))
            Text(
                if (crew.out) "Sent back" else "${crew.filled} of ${CrewRules.SEATS} in",
                style = tileStyle(12f, 17f, FontWeight.Normal),
                color = if (crew.out) Tone.BAD.ink() else Tt.colors.text3,
            )
        }
        if (!crew.out) {
            Spacer(Modifier.height(14.dp))
            CrewSeatMap(crew)
        }
        Spacer(Modifier.height(if (crew.out) 14.dp else 9.dp))
        LobbyStanding(crew)
        if (mine != null) {
            Spacer(Modifier.height(12.dp))
            Row(verticalAlignment = Alignment.CenterVertically) {
                AppMark(mine.packageName, mine.appName, mine.icon, size = 28.dp, ownIcon = true)
                Spacer(Modifier.width(8.dp))
                Text(
                    mine.appName,
                    style = tileStyle(13f, 18f, FontWeight.Normal),
                    color = Tt.colors.text2,
                    maxLines = 2,
                    overflow = TextOverflow.Ellipsis,
                )
            }
        }
    }
}

/** A dot identifies the owner's actual place, including a place still being checked. */
@Composable
internal fun CrewSeatMap(crew: Crew) {
    val confirmed = crew.filled.coerceIn(0, CrewRules.SEATS)
    val taken = crew.taken.coerceIn(confirmed, CrewRules.SEATS)
    val own =
        when (crew.mine?.state) {
            SeatState.CONFIRMED -> if (confirmed > 0) 0 else -1
            SeatState.CHECKING -> if (taken > confirmed) confirmed else -1
            else -> -1
        }
    Row(
        Modifier.fillMaxWidth().semantics {
            contentDescription =
                "$confirmed confirmed, ${taken - confirmed} checking, ${CrewRules.SEATS - taken} open seats." +
                if (own >= 0) " The dot marks your seat." else ""
        },
        horizontalArrangement = Arrangement.spacedBy(4.dp),
    ) {
        repeat(CrewRules.SEATS) { index ->
            val color =
                when {
                    index == own && index < confirmed -> Tt.colors.accentInk
                    index < confirmed -> Tt.colors.accentInk.copy(alpha = 0.62f)
                    index < taken -> Tone.WARN.ink().copy(alpha = 0.5f)
                    else -> Tt.colors.tile
                }
            Box(
                Modifier
                    .weight(1f)
                    .height(15.dp)
                    .clip(RoundedCornerShape(4.dp))
                    .background(color),
                contentAlignment = Alignment.Center,
            ) {
                if (index == own) Box(Modifier.size(4.dp).clip(CircleShape).background(Tt.colors.bg))
            }
        }
    }
}

@Composable
private fun LobbyStanding(crew: Crew) {
    val mine = crew.mine
    val checking = mine?.state == SeatState.CHECKING
    val reported = (mine?.reports ?: 0) > 0
    val headline =
        when {
            crew.out -> if (mine?.state == SeatState.REMOVED) "Taken out by the team" else "Your app needs a fix"
            checking -> "Members are checking your app"
            reported -> "${mine?.reports} of ${CrewRules.REPORTS_TO_RETURN} reports"
            !crew.filling -> if (crew.startsIn < 60_000L) "Starts any moment" else "Starts in ${untilStart(crew.startsIn)}"
            else -> "Your seat is held"
        }
    val detail =
        when {
            crew.out -> mine?.let(::lobbyRefund).orEmpty()
            checking -> "${mine?.yes} Yes, ${mine.no} No"
            reported -> "Check the report"
            !crew.filling -> if (crew.firstNight > 1) "Your day ${crew.firstNight}" else "At midnight"
            else -> "${(CrewRules.SEATS - crew.filled).coerceAtLeast(0)} more to start"
        }
    val tone =
        when {
            crew.out -> Tone.BAD
            checking || reported -> Tone.WARN
            else -> null
        }
    Row(Modifier.fillMaxWidth(), verticalAlignment = Alignment.Top) {
        Text(headline, Modifier.weight(1f), style = tileStyle(15f, 20f, FontWeight.Medium), color = tone?.ink() ?: Tt.colors.text)
        Spacer(Modifier.width(12.dp))
        Text(
            detail,
            Modifier.weight(1f),
            style = tileStyle(12f, 17f, FontWeight.Normal),
            color = Tt.colors.text3,
            textAlign = TextAlign.End,
        )
    }
}

/** Older servers omit ledger amounts; do not invent a refund while its result is unknown. */
private fun lobbyRefund(seat: CrewSeat): String =
    when {
        seat.refunded > 0 -> "${seat.refunded} tix returned"
        seat.charged == 0 -> "No tix charged"
        seat.refunded == 0 -> "No tix returned"
        else -> "See refund details"
    }

        }
    }
